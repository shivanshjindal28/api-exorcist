#   pip install -e ".[live]"            + semgrep, pyyaml  (scan real repos)
#   pip install -e ".[ml]"              + scikit-learn, shap  (model + SHAP)
#   pip install -e ".[graph]"           + neo4j driver
#   pip install -e ".[stream]"          + kafka, elasticsearch
#   pip install -e ".[dev]"             + pytest, ruff, mypy, pre-commit
#
# Everything for development:
#   pip install -e ".[dev,live,ml]"
#
# The core pipeline deliberately has no hard external dependencies: the demo,
# the test suite and CI must all run with no infrastructure and no network.
